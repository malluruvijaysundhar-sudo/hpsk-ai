<script>
// 1. PASTE YOUR GEMINI API KEY BELOW (Get one free from aistudio.google.com)
const GEMINI_API_KEY = "YOUR_GEMINI_API_KEY";

// 2. EDIT OR ADD ANY CUSTOM SCHOOL DETAILS HERE
const HPSK_CUSTOM_INFO = `
School Name: The Hyderabad Public School, Kadapa (HPSK)
Website: hpsk.org / hpskadapa.org
Established: 2005-2006 | Affiliation: CISCE (ICSE), New Delhi | Type: Day Boarding & Residential

Office Hours: Monday to Saturday, 9:00 AM to 5:00 PM
Facilities: Smart classrooms, science & computer labs, library, expansive sports grounds, separate hostels.
`;

let hpskChatHistory = [];

function toggleHpskWa() {
    const box = document.getElementById('hpsk-wa-box');
    box.style.display = (box.style.display === 'none' || box.style.display === '') ? 'flex' : 'none';
}

function quickHpskAsk(q) {
    document.getElementById('hpsk-wa-input').value = q;
    sendHpskWaMsg();
}

function formatWaReply(text) {
    return text
        .replace(/\*\*(.*?)\*\*/g, '<b>$1</b>')
        .replace(/\*(.*?)\*/g, '<i>$1</i>')
        .replace(/\n/g, '<br>');
}

async function sendHpskWaMsg() {
    const input = document.getElementById('hpsk-wa-input');
    const msgBox = document.getElementById('hpsk-wa-messages');
    const status = document.getElementById('hpsk-wa-status');
    const userText = input.value.trim();
    if (!userText) return;

    // Add user bubble
    msgBox.innerHTML += `<div style="background:#d9fdd3; color:#111; padding:8px 12px; border-radius:10px 0 10px 10px; max-width:80%; align-self:flex-end; box-shadow:0 1px 2px rgba(0,0,0,0.1); line-height:1.4;">${userText}</div>`;
    input.value = '';
    msgBox.scrollTop = msgBox.scrollHeight;
    status.innerText = "typing...";

    // Automatically grab all visible text on the current webpage
    const livePageContent = document.body.innerText.substring(0, 6000);

    const systemInstruction = `You are the official WhatsApp-style AI Assistant for The Hyderabad Public School, Kadapa (HPSK - hpsk.org).
Answer questions about HPSK (admissions, staff, facilities, events) AND general knowledge clearly, friendly, and concisely like WhatsApp Meta AI.
Use the custom school details below, the current webpage text, and Google Search when needed.

SCHOOL DETAILS:
${HPSK_CUSTOM_INFO}

CURRENT WEBPAGE CONTENT:
${livePageContent}`;

    hpskChatHistory.push({ role: "user", parts: [{ text: userText }] });

    try {
        const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=${GEMINI_API_KEY}`, {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({
                system_instruction: { parts: [{ text: systemInstruction }] },
                contents: hpskChatHistory,
                tools: [{ google_search: {} }]
            })
        });

        const data = await response.json();
        const aiReply = data?.candidates?.[0]?.content?.parts?.[0]?.text || "Please check that your Gemini API key is pasted correctly, or contact the school office at +91 85000 17606.";

        hpskChatHistory.push({ role: "model", parts: [{ text: aiReply }] });

        msgBox.innerHTML += `<div style="background:white; color:#111; padding:10px 12px; border-radius:0 10px 10px 10px; max-width:84%; align-self:flex-start; box-shadow:0 1px 2px rgba(0,0,0,0.1); line-height:1.4;">${formatWaReply(aiReply)}</div>`;
    } catch (error) {
        msgBox.innerHTML += `<div style="background:white; color:#111; padding:10px 12px; border-radius:0 10px 10px 10px; max-width:84%; align-self:flex-start;">Connection error. Please try again!</div>`;
    }

    status.innerText = "Online • Connected to Web & HPSK";
    msgBox.scrollTop = msgBox.scrollHeight;
}
</script>
